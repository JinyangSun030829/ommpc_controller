cmake_minimum_required(VERSION 3.0.2)
project(minisnap_offline_tests)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
add_library(minisnap_core STATIC ../src/trajectory_generator.cpp ../src/trajectory_planner.cpp)
target_include_directories(minisnap_core PUBLIC ../include ${EIGEN3_INCLUDE_DIRS})
add_executable(trajectory_generator_offline_test trajectory_generator_offline_test.cpp)
target_link_libraries(trajectory_generator_offline_test minisnap_core)
enable_testing()
add_executable(flight_safety_offline_test flight_safety_offline_test.cpp)
target_include_directories(flight_safety_offline_test PRIVATE ../include ${EIGEN3_INCLUDE_DIRS})
add_test(NAME flight_safety_offline COMMAND flight_safety_offline_test)
add_executable(hover_gate_offline_test hover_gate_offline_test.cpp)
target_include_directories(hover_gate_offline_test PRIVATE ../include ${EIGEN3_INCLUDE_DIRS})
add_test(NAME hover_gate_offline COMMAND hover_gate_offline_test)
add_test(NAME minisnap_offline COMMAND trajectory_generator_offline_test)
option(BUILD_ROS_NODE "Build node against installed ROS for isolated integration tests" OFF)
if(BUILD_ROS_NODE)
  find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs geometry_msgs std_msgs visualization_msgs mavros_msgs rosbag dynamic_reconfigure)
  find_package(Threads REQUIRED)
  add_executable(trajectory_generator_node ../src/trajectory_generator_node.cpp)
  target_include_directories(trajectory_generator_node PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(trajectory_generator_node minisnap_core ${catkin_LIBRARIES} ${CMAKE_THREAD_LIBS_INIT})
  add_executable(trajectory_manager_ros_test trajectory_manager_ros_test.cpp)
  target_include_directories(trajectory_manager_ros_test PRIVATE ../include ../include/traj_tracking_controller ${EIGEN3_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
  target_link_libraries(trajectory_manager_ros_test ${catkin_LIBRARIES})
  add_executable(traj_tracking_controller_fsm ../src/traj_tracking_controller_fsm.cpp)
  target_include_directories(traj_tracking_controller_fsm PRIVATE ../include ../include/traj_tracking_controller ${EIGEN3_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
  target_link_libraries(traj_tracking_controller_fsm ${catkin_LIBRARIES})
  add_executable(fsm_safety_ros_test fsm_safety_ros_test.cpp)
  target_include_directories(fsm_safety_ros_test PRIVATE ../include ../include/traj_tracking_controller ${EIGEN3_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
  target_link_libraries(fsm_safety_ros_test ${catkin_LIBRARIES})
endif()
